        package com.blitz.aimengine;

        import java.util.ArrayList;
        import java.util.Collections;
        import java.util.List;

        public class CarromAIPhysics {

            public static class Vector2 {
                public float x, y;
                public Vector2(float x, float y) { this.x = x; this.y = y; }
                public Vector2 add(Vector2 v) { return new Vector2(this.x + v.x, this.y + v.y); }
                public Vector2 sub(Vector2 v) { return new Vector2(this.x - v.x, this.y - v.y); }
                public Vector2 mul(float s) { return new Vector2(this.x * s, this.y * s); }
                public float length() { return (float) Math.hypot(x, y); }
                public Vector2 normalize() {
                    float len = length();
                    return len != 0 ? new Vector2(x / len, y / len) : new Vector2(0, 0);
                }
                public float dot(Vector2 v) { return this.x * v.x + this.y * v.y; }
            }

            public enum PuckType {
                RED_QUEEN, MY_PUCK, OPPONENT_PUCK
            }

            public static class Puck {
                public Vector2 pos;
                public PuckType type;
                public Puck(float x, float y, PuckType type) {
                    this.pos = new Vector2(x, y);
                    this.type = type;
                }
            }

            public static class AIShotPlan {
                public Vector2 strikerPos;
                public Vector2 strikerAimPoint;
                public Vector2 puckStart;
                public Vector2 targetPocket;
                public Vector2 cushionBounce;
                public PuckType targetType;
                public float score;
                public boolean isValid;
            }

            public static List<Vector2> getPockets(float w, float h) {
                List<Vector2> pockets = new ArrayList<>();
                pockets.add(new Vector2(100, 100));
                pockets.add(new Vector2(w - 100, 100));
                pockets.add(new Vector2(100, h - 100));
                pockets.add(new Vector2(w - 100, h - 100));
                return pockets;
            }

            public static AIShotPlan findBestAIShot(Vector2 striker, List<Puck> pucks, float w, float h) {
                List<AIShotPlan> shots = new ArrayList<>();
                List<Vector2> pockets = getPockets(w, h);

                for (Puck puck : pucks) {
                    if (puck.type == PuckType.OPPONENT_PUCK) continue;

                    for (Vector2 pocket : pockets) {
                        Vector2 puckToPocket = pocket.sub(puck.pos).normalize();
                        Vector2 impactPoint = puck.pos.sub(puckToPocket.mul(65f));
                        Vector2 strikerToImpact = impactPoint.sub(striker);

                        float alignment = strikerToImpact.normalize().dot(puckToPocket);
                        if (alignment > 0.15f) {
                            AIShotPlan plan = new AIShotPlan();
                            plan.strikerPos = striker;
                            plan.strikerAimPoint = impactPoint;
                            plan.puckStart = puck.pos;
                            plan.targetPocket = pocket;
                            plan.targetType = puck.type;
                            plan.isValid = true;

                            float dist = strikerToImpact.length() + puck.pos.sub(pocket).length();
                            float bonus = (puck.type == PuckType.RED_QUEEN) ? -3000f : 0f;
                            plan.score = dist + ((1.0f - alignment) * 600f) + bonus;

                            plan.cushionBounce = calculateBounce(pocket, puckToPocket, w, h);
                            shots.add(plan);
                        }
                    }
                }

                if (shots.isEmpty()) return null;
                Collections.sort(shots, (a, b) -> Float.compare(a.score, b.score));
                return shots.get(0);
            }

            private static Vector2 calculateBounce(Vector2 start, Vector2 dir, float w, float h) {
                Vector2 pos = new Vector2(start.x, start.y);
                Vector2 cur = dir.normalize();
                for (int i = 0; i < 2000; i++) {
                    pos.x += cur.x * 2.0f;
                    pos.y += cur.y * 2.0f;
                    if (pos.x <= 70 || pos.x >= w - 70) { cur.x = -cur.x; break; }
                    if (pos.y <= 70 || pos.y >= h - 70) { cur.y = -cur.y; break; }
                }
                return pos;
            }
        }

        package com.blitz.aimengine;

        import android.accessibilityservice.AccessibilityService;
        import android.accessibilityservice.GestureDescription;
        import android.graphics.Path;
        import android.os.Handler;
        import android.os.Looper;
        import android.view.accessibility.AccessibilityEvent;

        public class AutoPlayBotService extends AccessibilityService {

            private static AutoPlayBotService instance;

            public enum SpeedMode {
                HUMAN(350, 400),
                FAST(180, 200),
                ULTRA_FAST(70, 60);

                public final long aimDelay;
                public final long swipeDuration;
                SpeedMode(long delay, long dur) {
                    this.aimDelay = delay;
                    this.swipeDuration = dur;
                }
            }

            @Override
            public void onServiceConnected() {
                super.onServiceConnected();
                instance = this;
            }

            public static AutoPlayBotService getInstance() {
                return instance;
            }

            public void executeShot(float sX, float sY, float tX, float tY, SpeedMode mode) {
                if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.N) {
                    new Handler(Looper.getMainLooper()).postDelayed(() -> {
                        Path path = new Path();
                        path.moveTo(sX, sY);
                        path.lineTo(tX, tY);

                        GestureDescription.Builder builder = new GestureDescription.Builder();
                        builder.addStroke(new GestureDescription.StrokeDescription(path, 0, mode.swipeDuration));
                        dispatchGesture(builder.build(), null, null);
                    }, mode.aimDelay);
                }
            }

            @Override
            public void onAccessibilityEvent(AccessibilityEvent event) {}
            @Override
            public void onInterrupt() {}
        }

        package com.blitz.aimengine;

        import android.app.Service;
        import android.content.Intent;
        import android.graphics.*;
        import android.os.Build;
        import android.os.IBinder;
        import android.view.*;
        import android.widget.*;
        import java.util.ArrayList;
        import java.util.List;

        public class FloatingService extends Service {

            private WindowManager wm;
            private View canvasOverlay;
            private LinearLayout menuContainer;
            private TextView iconButton;

            private boolean showPrediction = true;
            private boolean predictionAfterShot = true;
            private boolean autoPlayEnabled = false;
            private AutoPlayBotService.SpeedMode currentSpeed = AutoPlayBotService.SpeedMode.HUMAN;

            private CarromAIPhysics.AIShotPlan currentPlan;

            @Override
            public void onCreate() {
                super.onCreate();
                wm = (WindowManager) getSystemService(WINDOW_SERVICE);

                int overlayType = Build.VERSION.SDK_INT >= Build.VERSION_CODES.O 
                        ? WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY 
                        : WindowManager.LayoutParams.TYPE_PHONE;

                // 1. Drawing Canvas for Prediction
                canvasOverlay = new View(this) {
                    Paint yellowGlow = new Paint();
                    Paint whiteLine = new Paint();
                    Paint redQueenLine = new Paint();
                    Paint reboundLine = new Paint();
                    Paint ghostRing = new Paint();

                    {
                        yellowGlow.setColor(Color.parseColor("#FACC15"));
                        yellowGlow.setStrokeWidth(5.5f);
                        yellowGlow.setStyle(Paint.Style.STROKE);
                        yellowGlow.setAntiAlias(true);

                        whiteLine.setColor(Color.WHITE);
                        whiteLine.setStrokeWidth(4.5f);
                        whiteLine.setStyle(Paint.Style.STROKE);
                        whiteLine.setAntiAlias(true);

                        redQueenLine.setColor(Color.parseColor("#EF4444"));
                        redQueenLine.setStrokeWidth(5f);
                        redQueenLine.setStyle(Paint.Style.STROKE);
                        redQueenLine.setAntiAlias(true);

                        reboundLine.setColor(Color.parseColor("#00E5FF"));
                        reboundLine.setStrokeWidth(4f);
                        reboundLine.setStyle(Paint.Style.STROKE);
                        reboundLine.setPathEffect(new DashPathEffect(new float[]{14, 8}, 0));
                        reboundLine.setAntiAlias(true);

                        ghostRing.setColor(Color.parseColor("#99FFFFFF"));
                        ghostRing.setStyle(Paint.Style.STROKE);
                        ghostRing.setStrokeWidth(3f);
                        ghostRing.setAntiAlias(true);
                    }

                    @Override
                    protected void onDraw(Canvas canvas) {
                        super.onDraw(canvas);
                        if (!showPrediction || currentPlan == null || !currentPlan.isValid) return;

                        // Striker Aim Ray
                        canvas.drawLine(currentPlan.strikerPos.x, currentPlan.strikerPos.y,
                                        currentPlan.strikerAimPoint.x, currentPlan.strikerAimPoint.y, yellowGlow);

                        // Impact Ghost Circle
                        canvas.drawCircle(currentPlan.strikerAimPoint.x, currentPlan.strikerAimPoint.y, 30f, ghostRing);

                        // Puck Path (White for My Puck, Red for Queen)
                        Paint targetLine = (currentPlan.targetType == CarromAIPhysics.PuckType.RED_QUEEN) ? redQueenLine : whiteLine;
                        canvas.drawLine(currentPlan.puckStart.x, currentPlan.puckStart.y,
                                        currentPlan.targetPocket.x, currentPlan.targetPocket.y, targetLine);

                        // Rebound
                        if (predictionAfterShot && currentPlan.cushionBounce != null) {
                            canvas.drawLine(currentPlan.targetPocket.x, currentPlan.targetPocket.y,
                                            currentPlan.cushionBounce.x, currentPlan.cushionBounce.y, reboundLine);
                        }
                    }
                };

                WindowManager.LayoutParams canvasParams = new WindowManager.LayoutParams(
                        WindowManager.LayoutParams.MATCH_PARENT, WindowManager.LayoutParams.MATCH_PARENT,
                        overlayType,
                        WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE | WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE | WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN,
                        PixelFormat.TRANSLUCENT
                );
                wm.addView(canvasOverlay, canvasParams);

                // 2. Setup Floating Trigger Icon & Menu
                buildFloatingMenu(overlayType);
                runAI();
            }

            private void buildFloatingMenu(int overlayType) {
                // Outer Layout Wrapper
                LinearLayout root = new LinearLayout(this);
                root.setOrientation(LinearLayout.VERTICAL);

                // Small App Logo / Icon button (Toggle menu)
                iconButton = new TextView(this);
                iconButton.setText(" ⚡ BLITZ ");
                iconButton.setBackgroundColor(Color.parseColor("#E60F172A"));
                iconButton.setTextColor(Color.parseColor("#FACC15"));
                iconButton.setPadding(25, 20, 25, 20);
                iconButton.setTextSize(14f);
                root.addView(iconButton);

                // Collapsible Menu Card
                menuContainer = new LinearLayout(this);
                menuContainer.setOrientation(LinearLayout.VERTICAL);
                menuContainer.setBackgroundColor(Color.parseColor("#F2111827"));
                menuContainer.setPadding(35, 30, 35, 30);
                menuContainer.setVisibility(View.GONE);

                // --- PREDICTION OPTION ---
                Switch swPrediction = new Switch(this);
                swPrediction.setText("Show Prediction");
                swPrediction.setTextColor(Color.WHITE);
                swPrediction.setChecked(true);

                LinearLayout subPrediction = new LinearLayout(this);
                subPrediction.setOrientation(LinearLayout.VERTICAL);
                subPrediction.setPadding(35, 8, 0, 10);

                CheckBox cbAfterShot = new CheckBox(this);
                cbAfterShot.setText("Prediction After Shot");
                cbAfterShot.setTextColor(Color.LTGRAY);
                cbAfterShot.setChecked(true);
                cbAfterShot.setOnCheckedChangeListener((b, isChecked) -> {
                    predictionAfterShot = isChecked;
                    canvasOverlay.invalidate();
                });
                subPrediction.addView(cbAfterShot);

                swPrediction.setOnCheckedChangeListener((b, isChecked) -> {
                    showPrediction = isChecked;
                    subPrediction.setVisibility(isChecked ? View.VISIBLE : View.GONE);
                    canvasOverlay.invalidate();
                });

                menuContainer.addView(swPrediction);
                menuContainer.addView(subPrediction);

                // --- AUTOPLAY OPTION ---
                Switch swAutoPlay = new Switch(this);
                swAutoPlay.setText("Enable AutoPlay");
                swAutoPlay.setTextColor(Color.parseColor("#38BDF8"));

                LinearLayout subAutoPlay = new LinearLayout(this);
                subAutoPlay.setOrientation(LinearLayout.VERTICAL);
                subAutoPlay.setPadding(35, 8, 0, 10);
                subAutoPlay.setVisibility(View.GONE);

                RadioGroup speedGroup = new RadioGroup(this);
                String[] modes = {"Human (Natural)", "Fast", "Ultra Fast (Instant)"};
                for (int i = 0; i < modes.length; i++) {
                    RadioButton rb = new RadioButton(this);
                    rb.setText(modes[i]);
                    rb.setTextColor(Color.WHITE);
                    rb.setId(i);
                    if (i == 0) rb.setChecked(true);
                    speedGroup.addView(rb);
                }

                speedGroup.setOnCheckedChangeListener((group, checkedId) -> {
                    if (checkedId == 0) currentSpeed = AutoPlayBotService.SpeedMode.HUMAN;
                    else if (checkedId == 1) currentSpeed = AutoPlayBotService.SpeedMode.FAST;
                    else currentSpeed = AutoPlayBotService.SpeedMode.ULTRA_FAST;
                });

                subAutoPlay.addView(speedGroup);

                swAutoPlay.setOnCheckedChangeListener((b, isChecked) -> {
                    autoPlayEnabled = isChecked;
                    subAutoPlay.setVisibility(isChecked ? View.VISIBLE : View.GONE);
                    if (isChecked) executeAIShot();
                });

                menuContainer.addView(swAutoPlay);
                menuContainer.addView(subAutoPlay);

                // Close Button
                Button btnClose = new Button(this);
                btnClose.setText("CLOSE ENGINE");
                btnClose.setBackgroundColor(Color.parseColor("#DC2626"));
                btnClose.setTextColor(Color.WHITE);
                btnClose.setOnClickListener(v -> stopSelf());
                menuContainer.addView(btnClose);

                root.addView(menuContainer);

                WindowManager.LayoutParams params = new WindowManager.LayoutParams(
                        WindowManager.LayoutParams.WRAP_CONTENT, WindowManager.LayoutParams.WRAP_CONTENT,
                        overlayType, WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE, PixelFormat.TRANSLUCENT
                );
                params.gravity = Gravity.TOP | Gravity.START;
                params.x = 40;
                params.y = 150;

                iconButton.setOnClickListener(v -> {
                    if (menuContainer.getVisibility() == View.VISIBLE) {
                        menuContainer.setVisibility(View.GONE);
                    } else {
                        menuContainer.setVisibility(View.VISIBLE);
                    }
                });

                // Draggable Menu
                root.setOnTouchListener(new View.OnTouchListener() {
                    private int initX, initY;
                    private float touchX, touchY;
                    @Override
                    public boolean onTouch(View v, MotionEvent e) {
                        switch (e.getAction()) {
                            case MotionEvent.ACTION_DOWN:
                                initX = params.x; initY = params.y;
                                touchX = e.getRawX(); touchY = e.getRawY();
                                return true;
                            case MotionEvent.ACTION_MOVE:
                                params.x = initX + (int)(e.getRawX() - touchX);
                                params.y = initY + (int)(e.getRawY() - touchY);
                                wm.updateViewLayout(root, params);
                                return true;
                        }
                        return false;
                    }
                });

                wm.addView(root, params);
            }

            private void runAI() {
                CarromAIPhysics.Vector2 striker = new CarromAIPhysics.Vector2(540, 1800);
                List<CarromAIPhysics.Puck> pucks = new ArrayList<>();
                pucks.add(new CarromAIPhysics.Puck(540, 1100, CarromAIPhysics.PuckType.RED_QUEEN)); // Red Queen Center
                pucks.add(new CarromAIPhysics.Puck(380, 1250, CarromAIPhysics.PuckType.MY_PUCK));
                pucks.add(new CarromAIPhysics.Puck(700, 1000, CarromAIPhysics.PuckType.OPPONENT_PUCK));

                currentPlan = CarromAIPhysics.findBestAIShot(striker, pucks, 1080, 2400);
                canvasOverlay.invalidate();
            }

            private void executeAIShot() {
                if (currentPlan != null && AutoPlayBotService.getInstance() != null) {
                    AutoPlayBotService.getInstance().executeShot(
                            currentPlan.strikerPos.x, currentPlan.strikerPos.y,
                            currentPlan.strikerAimPoint.x, currentPlan.strikerAimPoint.y,
                            currentSpeed
                    );
                }
            }

            @Override
            public void onDestroy() {
                super.onDestroy();
                if (canvasOverlay != null) wm.removeView(canvasOverlay);
            }

            @Override
            public IBinder onBind(Intent intent) { return null; }
        }

        package com.blitz.aimengine;

        import android.app.Activity;
        import android.content.Intent;
        import android.net.Uri;
        import android.os.Build;
        import android.os.Bundle;
        import android.provider.Settings;
        import android.widget.Toast;

        public class MainActivity extends Activity {
            @Override
            protected void onCreate(Bundle savedInstanceState) {
                super.onCreate(savedInstanceState);

                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M && !Settings.canDrawOverlays(this)) {
                    Intent intent = new Intent(
                        Settings.ACTION_MANAGE_OVERLAY_PERMISSION,
                        Uri.parse("package:" + getPackageName())
                    );
                    startActivityForResult(intent, 501);
                } else {
                    launchTool();
                }
            }

            private void launchTool() {
                startService(new Intent(this, FloatingService.class));
                Toast.makeText(this, "Blitz Pro Activated!", Toast.LENGTH_SHORT).show();
                finish();
            }

            @Override
            protected void onActivityResult(int req, int res, Intent data) {
                super.onActivityResult(req, res, data);
                if (req == 501 && Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                    if (Settings.canDrawOverlays(this)) {
                        launchTool();
                    }
                }
            }
        }
